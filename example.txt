    precondition
    {
        // No user inputs required
    }
    {
        // Create the base box of 20×30×40 millimeter
        fCuboid(context, id + "baseBox", {
            "corner1" : vector(0, 0, 0) * millimeter,
            "corner2" : vector(20, 30, 40) * millimeter
        });

        // Create a central hole cylinder (radius 10 millimeter, full height)
        fCylinder(context, id + "centralHole", {
            "bottomCenter" : vector(10, 15, 0) * millimeter,
            "topCenter"    : vector(10, 15, 40) * millimeter,
            "radius"       : 10 * millimeter
        });

        // Create four corner hole cylinders (radius 2 millimeter, full height)
        fCylinder(context, id + "cornerHole1", {
            "bottomCenter" : vector(2,  2,  0) * millimeter,
            "topCenter"    : vector(2,  2,  40) * millimeter,
            "radius"       : 2 * millimeter
        });
        fCylinder(context, id + "cornerHole2", {
            "bottomCenter" : vector(18, 2,  0) * millimeter,
            "topCenter"    : vector(18, 2,  40) * millimeter,
            "radius"       : 2 * millimeter
        });
        fCylinder(context, id + "cornerHole3", {
            "bottomCenter" : vector(18, 28, 0) * millimeter,
            "topCenter"    : vector(18, 28, 40) * millimeter,
            "radius"       : 2 * millimeter
        });
        fCylinder(context, id + "cornerHole4", {
            "bottomCenter" : vector(2,  28, 0) * millimeter,
            "topCenter"    : vector(2,  28, 40) * millimeter,
            "radius"       : 2 * millimeter
        });

        // Subtract the central hole from the base box
        opBoolean(context, id + "cutCenter", {
            "tools"         : qCreatedBy(id + "centralHole", EntityType.BODY),
            "targets"       : qCreatedBy(id + "baseBox",      EntityType.BODY),
            "operationType" : BooleanOperationType.SUBTRACTION
        });

        // Subtract each corner hole sequentially
        opBoolean(context, id + "cutCorner1", {
            "tools"         : qCreatedBy(id + "cornerHole1", EntityType.BODY),
            "targets"       : qCreatedBy(id + "baseBox",      EntityType.BODY),
            "operationType" : BooleanOperationType.SUBTRACTION
        });
        opBoolean(context, id + "cutCorner2", {
            "tools"         : qCreatedBy(id + "cornerHole2", EntityType.BODY),
            "targets"       : qCreatedBy(id + "baseBox",      EntityType.BODY),
            "operationType" : BooleanOperationType.SUBTRACTION
        });
        opBoolean(context, id + "cutCorner3", {
            "tools"         : qCreatedBy(id + "cornerHole3", EntityType.BODY),
            "targets"       : qCreatedBy(id + "baseBox",      EntityType.BODY),
            "operationType" : BooleanOperationType.SUBTRACTION
        });
        opBoolean(context, id + "cutCorner4", {
            "tools"         : qCreatedBy(id + "cornerHole4", EntityType.BODY),
            "targets"       : qCreatedBy(id + "baseBox",      EntityType.BODY),
            "operationType" : BooleanOperationType.SUBTRACTION
        });

    precondition
    {
        // If you want to expose the thickness parameter to the user, you can add it here:
        // annotation { "Name" : "Wall Thickness", "Default" : 1.7 * millimeter }
        // isParameter(definition.thickness, ThicknessType);
    }
    {
        // 1. Brick body
        fCuboid(context, id + "base_box", {
            "corner1": vector(0, 0, 0) * millimeter,
            "corner2": vector(31.8, 15.8, 9.6) * millimeter   // LEGO 2x4 dimensions
        });

        // 2. Create 1 stud and pattern 4x2
        fCylinder(context, id + "top_stud", {
            "bottomCenter": vector(4, 4, 9.6) * millimeter,
            "topCenter":    vector(4, 4, 11.3) * millimeter,  // 9.6 + stud height 1.7mm

        // Prepare transforms array and instance names for 8 studs

            }
        }
        opPattern(context, id + "stud_pattern", {
            "entities":                  qCreatedBy(id + "top_stud", EntityType.BODY),

        });
    });

annotation { "Feature Type Name" : "LEGO 4x4 Brick with Corner Cutout"}

    precondition
    {
    }
    {
        // Base brick body - 4x4 size
        fCuboid(context, id + "base_box", {
            "corner1": vector(0, 0, 0) * millimeter,
            "corner2": vector(31.8, 31.8, 9.6) * millimeter   // LEGO 4x4 dimensions
        });
        
        // Create corner cutout like the corner plate example
        fCuboid(context, id + "cornerCutout", {
            "corner1": vector(24, 24, 0) * millimeter,
            "corner2": vector(31.8, 31.8, 9.6) * millimeter
        });
        
        // Subtract corner
        opBoolean(context, id + "cutCorner", {
            "tools": qCreatedBy(id + "cornerCutout", EntityType.BODY),
            "targets": qCreatedBy(id + "base_box", EntityType.BODY),
            "operationType": BooleanOperationType.SUBTRACTION
        });

        // Create first stud
        fCylinder(context, id + "top_stud", {
            "bottomCenter": vector(4, 4, 9.6) * millimeter,
            "topCenter":    vector(4, 4, 11.3) * millimeter,  // 9.6 + stud height 1.7mm
            "radius":       2.4 * millimeter
        });

        // Pattern studs, skipping the cut corner
        var transforms = [];
        var names = [];
        for (var i = 0; i < 4; i += 1)
        {
            for (var j = 0; j < 4; j += 1)

                if (i >= 3 && j >= 3) continue;
                
                transforms = append(transforms,
                    transform(vector(i * 8, j * 8, 0) * millimeter)
                );
                names = append(names, "stud_" ~ i ~ "_" ~ j);
            }
        }
        
        opPattern(context, id + "stud_pattern", {
            "entities": qCreatedBy(id + "top_stud", EntityType.BODY),
            "transforms": transforms,
            "instanceNames": names,
            "copyPropertiesAndAttributes": true
        });
    });

            }
        }
        
        opPattern(context, id + "stud_pattern", {
            "entities": qCreatedBy(id + "top_stud", EntityType.BODY),
            "transforms": transforms,
            "instanceNames": names,
            "copyPropertiesAndAttributes": true
        });

            "radius": 0.8 * millimeter
        });
    });

#Example LEGO 2x3 L-Shaped Plate
FeatureScript 2625;
import(path : "onshape/std/geometry.fs", version : "2625.0");
import(path : "onshape/std/common.fs", version : "2625.0");

annotation { "Feature Type Name" : "LEGO 2x3 L-Shaped Plate" }
export const lego2x3LShapedPlate = defineFeature(function(context is Context, id is Id, definition is map)
    precondition
    {
    }
    {
        // Create L-shaped plate (combining LEGO and L-shape examples)
        // First horizontal leg
        fCuboid(context, id + "horizontalLeg", {
            "corner1": vector(0, 0, 0) * millimeter,
            "corner2": vector(23.7, 15.8, 3.2) * millimeter  // 3x2 dimensions
        });
        
        // Create cut block to make L-shape
        fCuboid(context, id + "cutBlock", {
            "corner1": vector(7.9, 7.9, 0) * millimeter,      // 1x1 position
            "corner2": vector(23.7, 15.8, 3.2) * millimeter   // to corner
        });
        
        // Subtract to create L-shape
        opBoolean(context, id + "cutL", {
            "tools": qCreatedBy(id + "cutBlock", EntityType.BODY),
            "targets": qCreatedBy(id + "horizontalLeg", EntityType.BODY),
            "operationType": BooleanOperationType.SUBTRACTION
        });
        
        // Create first stud
        fCylinder(context, id + "stud", {
            "bottomCenter": vector(4, 4, 3.2) * millimeter,
            "topCenter": vector(4, 4, 4.9) * millimeter,  // 3.2 + stud height 1.7mm
            "radius": 2.4 * millimeter
        });
        
        // Pattern studs, skipping the cutout area
        var transforms = [];
        var names = [];
        for (var i = 0; i < 3; i += 1)
        {
            for (var j = 0; j < 2; j += 1)
            {
                // Skip studs in the cut area
                if (i >= 1 && j >= 1) continue;
                
                transforms = append(transforms,
                    transform(vector(i * 8, j * 8, 0) * millimeter)
                );
                names = append(names, "stud_" ~ i ~ "_" ~ j);
            }
        }
        
        opPattern(context, id + "studPattern", {
            "entities": qCreatedBy(id + "stud", EntityType.BODY),
            "transforms": transforms,
            "instanceNames": names,
            "copyPropertiesAndAttributes": true
        });
    });
